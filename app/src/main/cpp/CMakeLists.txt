cmake_minimum_required(VERSION 3.22.1)
project(thor_whisper LANGUAGES C CXX)
include(FetchContent)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(whisper_cpp GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git GIT_TAG v1.7.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(whisper_cpp)
add_library(thor_whisper SHARED thor_whisper_jni.cpp)
target_compile_features(thor_whisper PRIVATE cxx_std_17)
target_link_libraries(thor_whisper PRIVATE whisper log android)
